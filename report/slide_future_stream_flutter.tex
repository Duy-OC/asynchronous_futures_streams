\begin{frame}{1.1 Vì sao cần lập trình bất đồng bộ?}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \small
      \begin{itemize}\itemsep4pt
        \item Dart chạy trên \textbf{một luồng (single-thread)} cho mỗi \textbf{isolate}.
        \item Tác vụ chờ: \textbf{mạng, đọc/ghi file, timer, cơ sở dữ liệu}\,\ldots{} nếu chặn luồng chính $\Rightarrow$ \textbf{UI bị đứng (jank)}.

          \item \textbf{Event Queue} (I/O, Timer, UI event)
        \end{enumerate}
        \item \textbf{Future} và \textbf{Stream} là hai công cụ cốt lõi của Dart để làm việc với Event Loop.
      \end{itemize}

\begin{frame}{1.1 Future: đại diện cho một kết quả ``sẽ có trong tương lai''}

        \item Tạo Future: \api{Future(...)}, \api{Future.delayed()}, \api{Future.value()}, \api{Future.error()}.

\begin{frame}[fragile]{1.1 Cú pháp \textbf{async/await}: viết bất đồng bộ như đồng bộ}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{Cách 1: \cd{then} / \cd{catchError}}
\begin{lstlisting}
Future<String> fetchUser() {
  return Future.delayed(
    Duration(seconds: 2),
    () => 'Nguyen Van A',
  );
}

void main() {
  fetchUser()
    .then((name) => print('Xin chao $name'))
    .catchError((e) => print('Loi: $e'))
    .whenComplete(() => print('Xong'));
}
\end{lstlisting}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{Cách 2: \cd{async} / \cd{await}} (khuyên dùng)
\begin{lstlisting}
Future<String> fetchUser() async {
  await Future.delayed(Duration(seconds: 2));
  return 'Nguyen Van A';
}

Future<void> main() async {
  try {
    final name = await fetchUser();
    print('Xin chao $name');
  } catch (e) {
    print('Loi: $e');
  } finally {
    print('Xong');
  }
}
\end{lstlisting}
    \end{column}
  \end{columns}
  \vspace{4pt}
  \small
  \begin{itemize}\itemsep2pt
    \item Hàm \textbf{async} luôn trả về \api{Future}; \textbf{await} chỉ dùng được \textbf{bên trong} hàm \textbf{async}.
    \item Code dễ đọc hơn, xử lý lỗi bằng \textbf{try/catch} quen thuộc, tránh ``callback hell''.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{1.2 \textbf{StreamController}: phát dữ liệu \& lắng nghe sự kiện}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
\begin{lstlisting}
final controller = StreamController<int>();

// Consumer: lang nghe (listener)
final sub = controller.stream.listen(
  (data) => print('Nhan: $data'),
  onError: (e) => print('Loi: $e'),
  onDone: () => print('Ket thuc'),
);

// Producer: phat du lieu
controller.add(1);
controller.add(2);
controller.addError(Exception('oops'));
controller.close();   // bao hieu "done"

// Hoac tao Stream bang async*
Stream<int> countDown(int n) async* {
  for (var i = n; i >= 0; i--) {
    await Future.delayed(Duration(seconds: 1));
    yield i;       // phat 1 gia tri
  }
}
\end{lstlisting}
    \end{column}
    \begin{column}{0.44\textwidth}
      \small
      \begin{itemize}\itemsep4pt
        \item \api{StreamController} là \textbf{nguồn phát}: \api{add()} gửi dữ liệu, \api{addError()} gửi lỗi, \api{close()} kết thúc.
        \item \api{controller.stream} là \textbf{đầu ra} để các listener đăng ký.
        \item \api{listen()} trả về \api{StreamSubscription} để \textbf{pause / resume / cancel}.
        \item \textbf{Single-subscription}: chỉ 1 listener (mặc định). \\ \textbf{Broadcast}: \api{StreamController.broadcast()} cho \textbf{nhiều} listener.
        \item \api{async*} + \api{yield}: cách \textbf{ngắn gọn} để tạo Stream.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

// pomodoro_timer.dart
(*@\texttt{\itshape // STREAM: phát toàn bộ trạng thái đồng hồ cho giao diện.}@*)
final StreamController<PomodoroState> _stateController =
    StreamController<PomodoroState>.broadcast();
Stream<PomodoroState> get stream => _stateController.stream;

void _startTicker() {
  _endTime = _now().add(_state.remaining);
  _ticker?.cancel();
  _ticker = Timer.periodic(tickInterval, (_) => _onTick());
  _emit(_state.copyWith(status: TimerStatus.running));
}

void _emit(PomodoroState nextState) {
  if (_disposed) return;
  _state = nextState;
  _stateController.add(nextState);   (*@\texttt{\itshape // phát cho listener}@*)
}

(*@\texttt{\itshape // Thoát màn hình: hủy nguồn phát trước, rồi đóng Stream.}@*)
void dispose() {
  if (_disposed) return;
  _disposed = true;
  _stopTicker();                     // _ticker?.cancel()

\section{Kết luận}

\begin{frame}{3. Kết luận}
  \small
  \begin{itemize}\itemsep5pt
    \item \textbf{Future} + \textbf{async/await}: xử lý tác vụ chờ cho \textbf{một kết quả} mà không chặn UI.
    \item \textbf{Stream} + \textbf{StreamController} + \textbf{StreamBuilder}: phát, lắng nghe và cập nhật giao diện theo \textbf{luồng dữ liệu liên tục}.
    \item \textbf{cancel() / close() / dispose()}: bắt buộc để tránh \textbf{memory leak}.
    \item Project \textbf{FocusFlow (Pomodoro)} áp dụng đủ ba kỹ thuật: \textbf{Stream} cho đồng hồ, \textbf{Future} cho lịch sử, \textbf{dispose} để dọn dẹp.
  \end{itemize}
\end{frame}
